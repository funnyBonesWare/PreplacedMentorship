The **`clearTimeout()`** method of the Window Interface (Representing Window Object) cancels a timeout previously established by calling window.setTimeout().

If the parameter provided does not identify a previously established action, this method does nothing.

```js
clearTimeout(timeoutID)
```

### <font color="#4bacc6">Example Usage</font>

```js
const alarm = {
  remind(aMessage) {
    alert(aMessage);
    this.timeoutID = undefined;
  },

  setup() {
    if (typeof this.timeoutID === "number") {
      this.cancel();
    }

    this.timeoutID = setTimeout(
      (msg) => {
        this.remind(msg);
      },
      1000,
      "Wake up!",
    );
  },

  cancel() {
    clearTimeout(this.timeoutID);
  },
};
window.addEventListener("click", () => alarm.setup());
```
